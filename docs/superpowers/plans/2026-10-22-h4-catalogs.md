~~~ts
export interface CriteriaAvailabilityInput {
  isActive: boolean;
  criteria: { maxScore: number; subCriteria: { options: unknown[] }[] }[];
}
export function assertCriteriaSetReady(set: CriteriaAvailabilityInput) {
  const maxScore = set.criteria.reduce((sum, criterion) => sum + criterion.maxScore, 0);
  const complete = set.criteria.length > 0 && set.criteria.every((criterion) =>
    criterion.subCriteria.length > 0 &&
    criterion.subCriteria.every((sub) => sub.options.length > 0));
  if (!set.isActive || !complete || maxScore !== 100) {
    throw new ConflictException('A verified active 100-point criteria set is required');
  }
}
~~~
